\chapter{نتیجه ممیزی و تغییرات انجام‌شده}

\section{نقاط قوت نسخه اولیه}
تنظیمات اصلی کلاک و پیرامونی‌ها روشن و خروجی ساخت \lr{Keil} موجود موفق و بدون هشدار بوده است. پروتکل ثابت، ضرایب کالیبراسیون عملیاتی، دو گذرگاه دما، شناسه‌های یکتا و هیسترزیس کنترل، دانش ارزشمند محصول را ثبت کرده‌اند. استفاده از نشانگرهای حفاظت‌شده \lr{CubeMX} در بخش‌هایی از فایل اصلی نیز پایه مناسبی بوده است.

\section{اشکال‌های بحرانی اصلاح‌شده}
\begin{itemize}
  \item عملیات طولانی \lr{SPI}، \lr{1-Wire} و \lr{UART} از متن وقفه خارج شد.
  \item خواندن مستقیم ثبات \lr{UART} پس از \lr{HAL} و فراخوانی دوباره هندلر حذف و دریافت استاندارد وقفه‌ای جایگزین شد.
  \item منطق کاربردی از فایل وقفه تولیدشده به ماژول مستقل انتقال یافت.
  \item اختلاف دوره تایمر دوم میان فایل مولد و سورس برطرف شد.
  \item خواندن از
\lr{0x08009000}
و نوشتن در
\lr{0x08008000}
با یک صفحه رزروشده، نسخه‌دار و دارای \lr{CRC} جایگزین شد.
  \item حلقه‌ای که ۶۴ مقدار را از آرایه ۳۲ عضوی می‌خواند حذف شد.
  \item خطر هم‌پوشانی داده تنظیمات با کد برنامه با کاهش ناحیه لینک برطرف شد.
  \item بلوک‌های بزرگ کد کامنت‌شده و متغیرهای بایت‌به‌بایت تکراری حذف و ساخت قاب حلقه‌ای شد.
  \item تقسیم بر صفر در محاسبه پنجره هیتر و خروجی خارج از بازه رطوبت مهار شد.
  \item حالت خاموش خودکار اکنون واقعاً پایدار و ایمن است.
\end{itemize}

\section{موارد حفظ‌شده}
قالب ۲۰۹ بایتی، نرخ سریال، نگاشت دو ایستگاه، ترتیب بایت، آدرس‌های پیش‌فرض، ضرایب کالیبراسیون، منطق فعال‌پایین، هیسترزیس و رابطه پنجره هیتر حفظ شده‌اند. جابه‌جایی ظاهری شمارنده‌های استارت نیز برای سازگاری قاب قدیمی حفظ و در کد روشن ثبت شده است.

\section{ریسک‌های باز}
\begin{enumerate}
  \item درایور \lr{1-Wire} هنوز زمان تبدیل کافی مطابق بدترین حالت دیتاشیت و \lr{CRC} داده را تضمین نمی‌کند.
  \item تبدیل‌های طولانی ۳۴ حسگر ترتیبی‌اند و می‌توانند نرخ حلقه را کاهش دهند.
  \item پروتکل ورودی همگام‌سازی، \lr{CRC}، پاسخ موفقیت و پاسخ خطا ندارد.
  \item حفاظت زمانی کمپرسور، تشخیص اتصال کوتاه حسگر و واچ‌داگ فعال نیست.
  \item نقش چند خروجی \lr{GPIO}، کانال دوم \lr{ADC} و دنباله بایت‌های ۱۷۷ تا ۲۰۴ مستند سخت‌افزاری ندارد.
  \item ضرایب انرژی و رطوبت تنها از کد استخراج شده‌اند و شناسنامه کالیبراسیون همراه پروژه نیست.
\end{enumerate}

\section{چرا بعضی فایل‌ها حذف نشدند}
کتابخانه‌های
\lr{AM2315}
و
\lr{MAF}
سورس مستقل و بالقوه مفیدند و حذفشان برای پروژه‌های بعدی برگشت‌پذیری را کم می‌کند؛ بنابراین با برچسب غیرفعال نگه داشته شدند. در مقابل، خروجی‌های ساخت قدیمی در این مرحله فیزیکی حذف نشدند، چون مسیر فعلی مخزن \lr{Git} نیست و ممکن است برای مقایسه باینری لازم باشند؛ فایل \lr{ignore} مانع ورود دوباره آن‌ها به کنترل نسخه خواهد شد.
